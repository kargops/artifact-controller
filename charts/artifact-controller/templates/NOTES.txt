1. Attach cloud identity, or every store check fails with StoreUnavailable:

     aws eks create-pod-identity-association \
       --cluster-name <cluster> \
       --namespace {{ .Release.Namespace }} \
       --service-account {{ include "artifact-controller.serviceAccountName" . }} \
       --role-arn arn:aws:iam::<account>:role/artifact-controller

{{- end }}

     - Tekton PipelineRuns
{{- end }}

     - Argo Workflows
{{- end }}
{{- if not (or .Values.generator.engines.batchJob .Values.generator.engines.tekton .Values.generator.engines.argoWorkflows) }}

     kubectl get artifactclasses
